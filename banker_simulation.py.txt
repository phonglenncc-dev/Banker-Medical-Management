class BankerAlgorithm:
    def __init__(self, processes, available, max_matrix, allocation_matrix):
        self.processes = processes  # Danh sách các Khoa/Phòng
        self.available = available  # Vector thiết bị rảnh rỗi hiện tại
        self.max = max_matrix       # Ma trận nhu cầu tối đa
        self.allocation = allocation_matrix  # Ma trận đang cấp phát
        
        # Tính ma trận Need = Max - Allocation
        self.need = [
            [self.max[i][j] - self.allocation[i][j] for j in range(len(available))]
            for i in range(len(processes))
        ]

    def check_safety(self):
        """Thuật toán kiểm tra trạng thái an toàn (Safety Algorithm)"""
        work = list(self.available)
        finish = [False] * len(self.processes)
        safe_sequence = []

        while len(safe_sequence) < len(self.processes):
            found = False
            for i in range(len(self.processes)):
                if not finish[i]:
                    # Kiểm tra xem Need của tiến trình có nhỏ hơn hoặc bằng Work không
                    if all(self.need[i][j] <= work[j] for j in range(len(work))):
                        # Giả định tiến trình hoàn thành và trả lại tài nguyên
                        for j in range(len(work)):
                            work[j] += self.allocation[i][j]
                        safe_sequence.append(self.processes[i])
                        finish[i] = True
                        found = True
                        break
            
            if not found:
                # Hệ thống không ở trạng thái an toàn (nguy cơ Deadlock)
                return False, []

        return True, safe_sequence

    def request_resources(self, process_index, request):
        """Xử lý yêu cầu cấp phát thiết bị phát sinh từ một khoa"""
        print(f"\n--- Khoa {self.processes[process_index]} yêu cầu thêm thiết bị: {request} ---")
        
        # 1. Kiểm tra Request <= Need
        if not all(request[j] <= self.need[process_index][j] for j in range(len(request))):
            print("Lỗi: Yêu cầu vượt quá nhu cầu tối đa đã đăng ký!")
            return False

        # 2. Kiểm tra Request <= Available
        if not all(request[j] <= self.available[j] for j in range(len(request))):
            print("Kho hiện không đủ thiết bị. Khoa phải chờ!")
            return False

        # 3. Giả lập cấp phát
        for j in range(len(request)):
            self.available[j] -= request[j]
            self.allocation[process_index][j] += request[j]
            self.need[process_index][j] -= request[j]

        # 4. Kiểm tra an toàn sau khi giả lập
        is_safe, sequence = self.check_safety()
        if is_safe:
            print("Trạng thái hệ thống: AN TOÀN. Phê duyệt yêu cầu cấp phát!")
            print(f"Chuỗi an toàn: {' -> '.join(sequence)}")
            return True
        else:
            print("Trạng thái hệ thống: KHÔNG AN TOÀN (Nguy cơ Deadlock). Từ chối cấp phát!")
            # Rollback (Hoàn tác trạng thái cũ)
            for j in range(len(request)):
                self.available[j] += request[j]
                self.allocation[process_index][j] -= request[j]
                self.need[process_index][j] += request[j]
            return False

    ]

        [3, 0, 2, 1],

    ]

    banker = BankerAlgorithm(processes, available, max_matrix, allocation_matrix)

    is_safe, seq = banker.check_safety()

        print(f"Hệ thống ban đầu AN TOÀN. Chuỗi an toàn: {' -> '.join(seq)}")

        print("Hệ thống ban đầu KHÔNG AN TOÀN!")
